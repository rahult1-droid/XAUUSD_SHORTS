name: XAUUSD Autonomous Shorts Pipeline

on:
  schedule:
    - cron: '0 6 * * *'  # Runs daily at 06:00 UTC automatically (zero laptop usage)
  workflow_dispatch:     # Allows you to manually trigger it anytime from the Actions tab

jobs:
  run-pipeline:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'

      - name: Install System Dependencies (FFmpeg)
        run: |
          sudo apt-get update
          sudo apt-get install -y ffmpeg

      - name: Install Python Dependencies
        run: |
          pip install google-genai edge-tts requests pydantic google-auth-oauthlib google-api-python-client

      - name: Execute Pipeline Script
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          YOUTUBE_CLIENT_ID: ${{ secrets.YOUTUBE_CLIENT_ID }}
          YOUTUBE_CLIENT_SECRET: ${{ secrets.YOUTUBE_CLIENT_SECRET }}
          YOUTUBE_REFRESH_TOKEN: ${{ secrets.YOUTUBE_REFRESH_TOKEN }}
        run: |
          python pipeline.py
